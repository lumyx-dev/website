import CodeTabs from '@/components/code-tabs.astro';

| Route | What it does |
|---|---|
| `GET /` | Built-in test client, mounted only when `SFU_SERVE_TEST_CLIENT` is on |
| `GET /ws` | WebSocket signaling, one connection per peer. See [Signaling protocol](/signaling) |
| `GET /health` | Status and running version |
| `GET /metrics` | Cumulative counters, as JSON. See [GET /metrics](/metrics) |
